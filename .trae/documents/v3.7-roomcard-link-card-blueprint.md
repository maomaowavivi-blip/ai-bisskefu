## 一、业务需求

客户在企微客服里发订单号(如 `5035035963841932965`),AI:
1. 查云房卡 (byChannelOrder)
2. 有云房卡 → 直接发 link 卡片给客户
3. 没有云房卡 → 先生成 (generateByChannelOrder) → 再发 link 卡片
4. 生成失败 → 降级发文本"稍后重试"

**link 卡片包含**:云房卡标题、shareText 描述、urlLink 链接、缩略图(临时 Unsplash 图)。

## 三、详细设计

### 3.1 新文件 `api/wecom_kf_roomcard_v37.php`

```php
<?php
declare(strict_types=1);

/**
 * 云房卡 link 卡片生成与发送(v3.7)
 *
 * 流程:
 * 1. 调 byChannelOrder 查云房卡
 * 2. 拿到 share_bundle,解析 urlLink / shareText
 * 3. 没卡 → 调 generateByChannelOrder 生成,再解析
 * 4. 构造 link 卡片参数,调用 wcom_kf.php 里的 sendKfLinkMessage
 */

require_once __DIR__ . '/config.php';

// 临时缩略图 URL(Unsplash 公开图片)
define('ROOMCARD_THUMB_URL', 'https://images.unsplash.com/photo-1631049307264-da0ec9d70304?w=520&h=416&fit=crop&q=80');

// 缓存:每个订单号对应的 thumb_media_id(避免每次重复上传)
function getThumbMediaIdCached(PDO $db): string {
    // 企微素材库上传(image 类型),返回 media_id
    // 缓存到 DB 或文件,首次上传后永久使用
    $cacheFile = __DIR__ . '/../logs/roomcard_thumb_media_id.txt';
    if (file_exists($cacheFile)) {
        $cached = trim(file_get_contents($cacheFile));
        if ($cached !== '') return $cached;
    }

    $mediaId = uploadImageToWeCom($db, ROOMCARD_THUMB_URL);
    if ($mediaId) {
        @file_put_contents($cacheFile, $mediaId);
    }
    return $mediaId;
}

/**
 * 调企微上传临时素材接口
 */
function uploadImageToWeCom(PDO $db, string $imgUrl): ?string {
    $corpId = trim(pcGet($db, 'wecom.corpid', ''));
    $secret = trim(pcGet($db, 'wecom.corp_secret', ''));
    $token = getAccessToken(); // 复用 wcom_kf.php 的 access_token
    if (!$token) return null;

    // 1. 下载图片到本地临时文件
    $tmpFile = tempnam(sys_get_temp_dir(), 'thumb_');
    $ch = curl_init($imgUrl);
    curl_setopt_array($ch, [CURLOPT_RETURNTRANSFER => true, CURLOPT_FOLLOWLOCATION => true, CURLOPT_TIMEOUT => 10]);
    $imgData = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    if ($httpCode !== 200 || !$imgData) return null;
    file_put_contents($tmpFile, $imgData);

    // 2. 上传到企微素材库
    $url = "https://qyapi.weixin.qq.com/cgi-bin/media/upload?access_token=$token&type=image";
    $postFields = ['media' => new CURLFile($tmpFile)];
    $ch = curl_init($url);
    curl_setopt_array($ch, [
        CURLOPT_POST => true,
        CURLOPT_POSTFIELDS => $postFields,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_TIMEOUT => 30,
    ]);
    $resp = curl_exec($ch);
    curl_close($ch);
    @unlink($tmpFile);

    $data = json_decode($resp, true);
    return $data['media_id'] ?? null;
}

/**
 * 查云房卡(根据 channel_order_id)
 */
function getRoomCard(PDO $db, string $channelOrderId): ?array {
    $user = trim(pcGet($db, 'roomcard.username', '13768370505'));
    $pwd  = trim(pcGet($db, 'roomcard.password', '7s9Kp2%26qG5'));
    $url = "https://apicenter.sujia365.com/index.php/openapi/room_card/byChannelOrder"
        . "?username=" . urlencode($user)
        . "&pwd=" . urlencode($pwd)
        . "&channel_order_id=" . urlencode($channelOrderId);

    $ch = curl_init($url);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_TIMEOUT => 10,
    ]);
    $resp = curl_exec($ch);
    curl_close($ch);

    $data = json_decode($resp, true);
    if (!isset($data['code']) || intval($data['code']) !== 1 || empty($data['data']['list'])) {
        return null;
    }
    return $data['data']['list'][0]; // 取第一个卡
}

/**
 * 生成云房卡(根据 channel_order_id)
 */
function generateRoomCard(PDO $db, string $channelOrderId): ?array {
    $user = trim(pcGet($db, 'roomcard.username', '13768370505'));
    $pwd  = trim(pcGet($db, 'roomcard.password', '7s9Kp2%26qG5'));
    $url = "https://apicenter.sujia365.com/index.php/openapi/room_card/generateByChannelOrder"
        . "?username=" . urlencode($user)
        . "&pwd=" . urlencode($pwd)
        . "&channel_order_id=" . urlencode($channelOrderId);

    $ch = curl_init($url);
    curl_setopt_array($ch, [
        CURLOPT_POST => true,
        CURLOPT_POSTFIELDS => [],  // POST 但无 body
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_TIMEOUT => 15,
    ]);
    $resp = curl_exec($ch);
    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);

    if ($httpCode !== 200) return null;
    $data = json_decode($resp, true);
    if (!isset($data['code']) || intval($data['code']) !== 1) return null;
    if (empty($data['data']['cards'])) return null;
    return $data['data']['cards'][0];
}

/**
 * 从云房卡记录里提取 urlLink 和 shareText
 */
function parseRoomCardForLink(array $card): ?array {
    // 优先从 share_bundle 解析
    $bundleRaw = $card['share_bundle'] ?? '';
    if (!empty($bundleRaw)) {
        $bundle = json_decode($bundleRaw, true);
        if (isset($bundle['miniapps'][0])) {
            $mini = $bundle['miniapps'][0];
            return [
                'title' => $mini['name'] ?? '宿家云房卡',
                'desc'  => $bundle['shareText'] ?? '点击查看您的云房卡',
                'url'   => $mini['urlLink'] ?? '',
            ];
        }
    }

    // share_bundle 为空时降级:用 urlLink 字段(如果有)
    if (!empty($card['urlLink'])) {
        return [
            'title' => '宿家云房卡',
            'desc'  => '点击查看您的云房卡',
            'url'   => $card['urlLink'],
        ];
    }

    return null;
}

/**
 * 主入口:返回 link 卡片参数数组
 */
function buildRoomCardLink(PDO $db, string $channelOrderId): ?array {
    $card = getRoomCard($db, $channelOrderId);
    if (!$card) {
        // 没卡,生成
        $card = generateRoomCard($db, $channelOrderId);
        if (!$card) return null;
    }

    $parsed = parseRoomCardForLink($card);
    if (!$parsed || empty($parsed['url'])) return null;

    $thumb = getThumbMediaIdCached($db);
    if (!$thumb) return null;

    return [
        'title' => $parsed['title'],
        'desc'  => $parsed['desc'],
        'url'   => $parsed['url'],
        'thumb_media_id' => $thumb,
    ];
}
```

### 3.2 修改 `api/wecom_kf.php`

**位置**:`sendKfMessage()` 后面加 `sendKfLinkMessage()`

```php
function sendKfLinkMessage(string $externalUserId, array $link, string $useOpenKfId = ''): bool {
    $accessToken = getAccessToken();
    if ($accessToken === null) return false;

    $openKfId = $useOpenKfId !== '' ? $useOpenKfId : getConfiguredOpenKfId();
    $url = "https://qyapi.weixin.qq.com/cgi-bin/kf/send_msg?access_token=" . urlencode($accessToken);

    $payload = [
        'touser' => $externalUserId,
        'open_kfid' => $openKfId,
        'msgtype' => 'link',
        'link' => [
            'title' => mb_substr($link['title'], 0, 128),
            'desc'  => mb_substr($link['desc'] ?? '', 0, 512),
            'url'   => $link['url'],
            'thumb_media_id' => $link['thumb_media_id'] ?? '',
        ],
    ];

    $resp = httpPostJson($url, $payload);
    if (!$resp || !isset($resp['errcode'])) return false;
    if ($resp['errcode'] !== 0) {
        wecom_kf_log('send_kf_link error: ' . json_encode($resp, JSON_UNESCAPED_UNICODE));
        return false;
    }
    wecom_kf_log("Sent kf link to $externalUserId: " . mb_substr($link['title'], 0, 30));
    return true;
}
```

**位置**:`processKfEvent()` 里,检测到 ORDER_QUERY Intent 后,先调 `buildRoomCardLink()` 返回 link 卡片参数(若有),发 link 卡片,不再走兜底话术
